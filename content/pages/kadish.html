Yisgadal v'yiskadash sh'mei rabo. (Omein)

בְּעָלְ֒מָֽא דִּי־בְרָא כִרְעוּתֵהּ וְיַמְלִיךְ מַלְכוּתֵהּ בְּחַיֵּיכוֹן וּבְיוֹמֵיכוֹן וּבְחַיֵּי דְכָל־בֵּית יִשְׂרָאֵל בַּעֲגָלָא וּבִזְמַן קָרִיב וְאִמְרוּ אָמֵן:

B'olmo di v'ro chir'usei, v'yamlich malchusei, b'chayeichon uv'yomeichon uv'chayei d'chol beis Yisroel, ba'agolo uvizman koriv, v'imru Omein.

יְהֵא שְׁמֵהּ רַבָּא מְבָרַךְ לְעָלַם וּלְעָלְ֒מֵי עָלְ֒מַיָּא

Y'hei sh'mei rabo m'vorach l'olam ul'olmei olmayo.

יִתְבָּרַךְ וְיִשְׁתַּבַּח וְיִתְפָּאַר וְיִתְרוֹמַם וְיִתְנַשֵּׂא וְיִתְהַדָּר וְיִתְעַלֶּה וְיִתְהַלָּל שְׁמֵהּ דְקוּדְשָׁא בְּרִיךְ הוּא

Yisborach v'yishtabach v'yispo'ar v'yisromam v'yisnasei v'yis'hadar v'yis'aleh v'yis'halol sh'mei d'kudsho b'rich hu.

לְעֵֽלָּא מִן־כָּל־ (בעשי"ת לְעֵֽלָּא לְעֵֽלָּא מִכָּל )בִּרְכָתָֽא וְשִׁירָתָֽא תֻּשְׁבְּ֒חָתָֽא וְנֶחָמָתָֽא דַּאֲמִירָן בְּעָלְ֒מָֽא וְאִמְרוּ אָמֵן:

L'eilo min kol (L'eilo l'eilo mikol) birchoso v'shiroso tushb'choso v'nechemoso, da'amiron b'olmo, v'imru Omein.

יְהֵא שְׁלָמָֽא רַבָּֽא מִן־שְׁמַיָּֽא וְחַיִּים עָלֵֽינוּ וְעַל־כָּל־יִשְׂרָאֵל וְאִמְרוּ אָמֵן:

Y'hei sh'lomo rabo min sh'mayo, v'chayim oleinu v'al kol Yisroel, v'imru Omein.

עוֹשֶׂה שָׁלוֹם (בעשי"ת הַשָּׁלוֹם) בִּמְרוֹמָיו הוּא יַעֲשֶׂה שָׁלוֹם עָלֵֽינוּ וְעַל כָּל יִשְׂרָאֵל וְאִמְרוּ אָמֵן:

אל מלא רחמים

לזכר

קֵל מָלֵא רַחֲמִים, שׁוֹכֵן בַּמְּרוֹמִים, הַמְצֵא מְנוּחָה נְכוֹנָה עַל כַּנְפֵי הַשְּׁכִינָה, בְּמַעֲלוֹת קְדוֹשִׁים וּטְהוֹרִים, כְּזֹֽהַר הָרָקִֽיעַ מַזְהִירִים, לְנִשְׁמַת (פלוני בֶּן פלוני) שֶׁהָלַךְ לְעוֹלָמוֹ, בַּעֲבוּר שֶׁבְּלִי נֶֽדֶר אֶתֵּן צְדָקָה בְּעַד הַזְכָּרַת נִשְׁמָתוֹ, בְּגַן עֵֽדֶן תְּהֵא מְנוּחָתוֹ. לָכֵן, בַּֽעַל הָרַחֲמִים יַסְתִּירֵֽהוּ בְּסֵֽתֶר כְּנָפָיו לְעוֹלָמִים, וְיִצְרֹר בִּצְרוֹר הַחַיִּים אֶת נִשְׁמָתוֹ, ד' הוּא נַחֲלָתוֹ, וְיָנֽוּחַ בְּשָׁלוֹם עַל מִשְׁכָּבוֹ, וְנֹאמַר אָמֵן.

לנקבה

בבית הקברות

מי שלא ראה קברי ישראל שלושים יום, מברך כשהוא בא סמוך לקברים:

בָּרוּךְ אַתָּה ד' אֱלֹקֵֽינוּ מֶֽלֶךְ הָעוֹלָם, אֲשֶׁר יָצַר אֶתְכֶם בַּדִּין, וְזָן וְכִלְכֵּל אֶתְכֶם בַּדִּין, וְהֵמִית אֶתְכֶם בַּדִּין, וְיוֹדֵֽעַ מִסְפַּר כֻּלְּכֶם בַּדִּין, וְהוּא עָתִיד לְהַחֲיוֹתְכֶם וּלְקַיֵּם אֶתְכֶם בַּדִּין. בָּרוּךְ אַתָּה ד', מְחַיֵּה הַמֵּתִים.

אַתָּה גִּבּוֹר לְעוֹלָם א', מְחַיֵּה מֵתִים אַתָּה, רַב לְהוֹשִֽׁיעַ, מְכַלְכֵּל חַיִּים בְּחֶֽסֶד, מְחַיֵּה מֵתִים בְּרַחֲמִים רַבִּים, סוֹמֵךְ נוֹפְלִים, וְרוֹפֵא חוֹלִים, וּמַתִּיר אֲסוּרִים, וּמְקַיֵּם אֱמוּנָתוֹ לִישֵׁנֵי עָפָר. מִי כָמֽוֹךָ בַּֽעַל גְּבוּרוֹת וּמִי דּֽוֹמֶה לָּךְ, מֶֽלֶךְ מֵמִית וּמְחַיֶּה וּמַצְמִֽיחַ יְשׁוּעָה, וְנֶאֱמָן אַתָּה לְהַחֲיוֹת מֵתִים.

תהלים י״ז

תְּפִלָּה לְדָוִד שִׁמְעָה ד' צֶדֶק הַקְשִׁיבָה רִנָּתִי הַאֲזִינָה תְפִלָּתִי בְּלֹא שִׂפְתֵי מִרְמָה׃ מִלְּפָנֶיךָ מִשְׁפָּטִי יֵצֵא עֵינֶיךָ תֶּחֱזֶינָה מֵישָׁרִים׃ בָּחַנְתָּ לִבִּי פָּקַדְתָּ לַּיְלָה צְרַפְתַּנִי בַל־תִּמְצָא זַמֹּתִי בַּל־יַעֲבָר־פִּי׃ לִפְעֻלּוֹת אָדָם בִּדְבַר שְׂפָתֶיךָ אֲנִי שָׁמַרְתִּי אָרְחוֹת פָּרִיץ׃ תָּמֹךְ אֲשֻׁרַי בְּמַעְגְּלוֹתֶיךָ בַּל־נָמוֹטּוּ פְעָמָי׃ אֲנִי־קְרָאתִיךָ כִי־תַעֲנֵנִי קֵל הַט־אָזְנְךָ לִי שְׁמַע אִמְרָתִי׃ הַפְלֵה חֲסָדֶיךָ מוֹשִׁיעַ חוֹסִים מִמִּתְקוֹמְמִים בִּימִינֶךָ׃ שָׁמְרֵנִי כְּאִישׁוֹן בַּת־עָיִן בְּצֵל כְּנָפֶיךָ תַּסְתִּירֵנִי׃ מִפְּנֵי רְשָׁעִים זוּ שַׁדּוּנִי אֹיְבַי בְּנֶפֶשׁ יַקִּיפוּ עָלָי׃ חֶלְבָּמוֹ סָּגְרוּ פִּימוֹ דִּבְּרוּ בְגֵאוּת׃ אַשֻּׁרֵינוּ עַתָּה סבבוני [סְבָבוּנוּ] עֵינֵיהֶם יָשִׁיתוּ לִנְטוֹת בָּאָרֶץ׃ דִּמְיֹנוֹ כְּאַרְיֵה יִכְסוֹף לִטְרוֹף וְכִכְפִיר יֹשֵׁב בְּמִסְתָּרִים׃ קוּמָה ד' קַדְּמָה פָנָיו הַכְרִיעֵהוּ פַּלְּטָה נַפְשִׁי מֵרָשָׁע חַרְבֶּךָ׃ מִמְתִים יָדְךָ ד' מִמְתִים מֵחֶלֶד חֶלְקָם בַּחַיִּים וצפינך [וּצְפוּנְךָ] תְּמַלֵּא בִטְנָם יִשְׂבְּעוּ בָנִים וְהִנִּיחוּ יִתְרָם לְעוֹלְלֵיהֶם׃ אֲנִי בְּצֶדֶק אֶחֱזֶה פָנֶיךָ אֶשְׂבְּעָה בְהָקִיץ תְּמוּנָתֶךָ׃

תהלים ע״ב

לִשְׁלֹמֹה אֱלֹקִים מִשְׁפָּטֶיךָ לְמֶלֶךְ תֵּן וְצִדְקָתְךָ לְבֶן־מֶלֶךְ׃ יָדִין עַמְּךָ בְצֶדֶק וַעֲנִיֶּיךָ בְמִשְׁפָּט׃ יִשְׂאוּ הָרִים שָׁלוֹם לָעָם וּגְבָעוֹת בִּצְדָקָה׃ יִשְׁפֹּט עֲנִיֵּי־עָם יוֹשִׁיעַ לִבְנֵי אֶבְיוֹן וִידַכֵּא עוֹשֵׁק׃ יִירָאוּךָ עִם־שָׁמֶשׁ וְלִפְנֵי יָרֵחַ דּוֹר דּוֹרִים׃ יֵרֵד כְּמָטָר עַל־גֵּז כִּרְבִיבִים זַרְזִיף אָרֶץ׃ יִפְרַח־בְּיָמָיו צַדִּיק וְרֹב שָׁלוֹם עַד־בְּלִי יָרֵחַ׃ וְיֵרְדְּ מִיָּם עַד־יָם וּמִנָּהָר עַד־אַפְסֵי־אָרֶץ׃ לְפָנָיו יִכְרְעוּ צִיִּים וְאֹיְבָיו עָפָר יְלַחֵכוּ׃ מַלְכֵי תַרְשִׁישׁ וְאִיִּים מִנְחָה יָשִׁיבוּ מַלְכֵי שְׁבָא וּסְבָא אֶשְׁכָּר יַקְרִיבוּ׃ וְיִשְׁתַּחֲווּ־לוֹ כָל־מְלָכִים כָּל־גּוֹיִם יַעַבְדוּהוּ׃ כִּי־יַצִּיל אֶבְיוֹן מְשַׁוֵּעַ וְעָנִי וְאֵין־עֹזֵר לוֹ׃ יָחֹס עַל־דַּל וְאֶבְיוֹן וְנַפְשׁוֹת אֶבְיוֹנִים יוֹשִׁיעַ׃ מִתּוֹךְ וּמֵחָמָס יִגְאַל נַפְשָׁם וְיֵיקַר דָּמָם בְּעֵינָיו׃ וִיחִי וְיִתֶּן־לוֹ מִזְּהַב שְׁבָא וְיִתְפַּלֵּל בַּעֲדוֹ תָמִיד כָּל־הַיּוֹם יְבָרֲכֶנְהוּ׃ יְהִי פִסַּת־בַּר בָּאָרֶץ בְּרֹאשׁ הָרִים יִרְעַשׁ כַּלְּבָנוֹן פִּרְיוֹ וְיָצִיצוּ מֵעִיר כְּעֵשֶׂב הָאָרֶץ׃ יְהִי שְׁמוֹ לְעוֹלָם לִפְנֵי־שֶׁמֶשׁ ינין [יִנּוֹן] שְׁמוֹ וְיִתְבָּרְכוּ בוֹ כָּל־גּוֹיִם יְאַשְּׁרוּהוּ׃ בָּרוּךְ ד' אֱלֹקִים אֱלֹקֵי יִשְׂרָאֵל עֹשֵׂה נִפְלָאוֹת לְבַדּוֹ׃ וּבָרוּךְ שֵׁם כְּבוֹדוֹ לְעוֹלָם וְיִמָּלֵא כְבוֹדוֹ אֶת־כֹּל הָאָרֶץ אָמֵן וְאָמֵן׃ כָּלּוּ תְפִלּוֹת דָּוִד בֶּן־יִשָׁי׃

תהלים ק״ד

בָּרֲכִי נַפְשִׁי אֶת־ד' ד' אֱלֹקַי גָּדַלְתָּ מְּאֹד הוֹד וְהָדָר לָבָשְׁתָּ׃ עֹטֶה־אוֹר כַּשַּׂלְמָה נוֹטֶה שָׁמַיִם כַּיְרִיעָה׃ הַמְקָרֶה בַמַּיִם עֲלִיּוֹתָיו הַשָּׂם־עָבִים רְכוּבוֹ הַמְהַלֵּךְ עַל־כַּנְפֵי־רוּחַ׃ עֹשֶׂה מַלְאָכָיו רוּחוֹת מְשָׁרְתָיו אֵשׁ לֹהֵט׃ יָסַד־אֶרֶץ עַל־מְכוֹנֶיהָ בַּל־תִּמּוֹט עוֹלָם וָעֶד׃ תְּהוֹם כַּלְּבוּשׁ כִּסִּיתוֹ עַל־הָרִים יַעַמְדוּ־מָיִם׃ מִן־גַּעֲרָתְךָ יְנוּסוּן מִן־קוֹל רַעַמְךָ יֵחָפֵזוּן׃ יַעֲלוּ הָרִים יֵרְדוּ בְקָעוֹת אֶל־מְקוֹם זֶה יָסַדְתָּ לָהֶם׃ גְּבוּל־שַׂמְתָּ בַּל־יַעֲבֹרוּן בַּל־יְשׁוּבוּן לְכַסּוֹת הָאָרֶץ׃ הַמְשַׁלֵּחַ מַעְיָנִים בַּנְּחָלִים בֵּין הָרִים יְהַלֵּכוּן׃ יַשְׁקוּ כָּל־חַיְתוֹ שָׂדָי יִשְׁבְּרוּ פְרָאִים צְמָאָם׃ עֲלֵיהֶם עוֹף־הַשָּׁמַיִם יִשְׁכּוֹן מִבֵּין עֳפָאיִם יִתְּנוּ־קוֹל׃ מַשְׁקֶה הָרִים מֵעֲלִיּוֹתָיו מִפְּרִי מַעֲשֶׂיךָ תִּשְׂבַּע הָאָרֶץ׃ מַצְמִיחַ חָצִיר לַבְּהֵמָה וְעֵשֶׂב לַעֲבֹדַת הָאָדָם לְהוֹצִיא לֶחֶם מִן־הָאָרֶץ׃ וְיַיִן יְשַׂמַּח לְבַב־אֱנוֹשׁ לְהַצְהִיל פָּנִים מִשָּׁמֶן וְלֶחֶם לְבַב־אֱנוֹשׁ יִסְעָד׃ יִשְׂבְּעוּ עֲצֵי ד' אַרְזֵי לְבָנוֹן אֲשֶׁר נָטָע׃ אֲשֶׁר־שָׁם צִפֳּרִים יְקַנֵּנוּ חֲסִידָה בְּרוֹשִׁים בֵּיתָהּ׃ הָרִים הַגְּבֹהִים לַיְּעֵלִים סְלָעִים מַחְסֶה לַשְׁפַנִּים׃ עָשָׂה יָרֵחַ לְמוֹעֲדִים שֶׁמֶשׁ יָדַע מְבוֹאוֹ׃ תָּשֶׁת־חֹשֶׁךְ וִיהִי לָיְלָה בּוֹ־תִרְמֹשׂ כָּל־חַיְתוֹ־יָעַר׃ הַכְּפִירִים שֹׁאֲגִים לַטָּרֶף וּלְבַקֵּשׁ מֵקֵל אָכְלָם׃ תִּזְרַח הַשֶּׁמֶשׁ יֵאָסֵפוּן וְאֶל־מְעוֹנֹתָם יִרְבָּצוּן׃ יֵצֵא אָדָם לְפָעֳלוֹ וְלַעֲבֹדָתוֹ עֲדֵי־עָרֶב׃ מָה־רַבּוּ מַעֲשֶׂיךָ ד' כֻּלָּם בְּחָכְמָה עָשִׂיתָ מָלְאָה הָאָרֶץ קִנְיָנֶךָ׃ זֶה הַיָּם גָּדוֹל וּרְחַב יָדָיִם שָׁם־רֶמֶשׂ וְאֵין מִסְפָּר חַיּוֹת קְטַנּוֹת עִם־גְּדֹלוֹת׃ שָׁם אֳנִיּוֹת יְהַלֵּכוּן לִוְיָתָן זֶה־יָצַרְתָּ לְשַׂחֶק־בּוֹ׃ כֻּלָּם אֵלֶיךָ יְשַׂבֵּרוּן לָתֵת אָכְלָם בְּעִתּוֹ׃ תִּתֵּן לָהֶם יִלְקֹטוּן תִּפְתַּח יָדְךָ יִשְׂבְּעוּן טוֹב׃ תַּסְתִּיר פָּנֶיךָ יִבָּהֵלוּן תֹּסֵף רוּחָם יִגְוָעוּן וְאֶל־עֲפָרָם יְשׁוּבוּן׃ תְּשַׁלַּח רוּחֲךָ יִבָּרֵאוּן וּתְחַדֵּשׁ פְּנֵי אֲדָמָה׃ יְהִי כְבוֹד ד' לְעוֹלָם יִשְׂמַח ד' בְּמַעֲשָׂיו׃ הַמַּבִּיט לָאָרֶץ וַתִּרְעָד יִגַּע בֶּהָרִים וְיֶעֱשָׁנוּ׃ אָשִׁירָה לַד' בְּחַיָּי אֲזַמְּרָה לֵאלֹקַי בְּעוֹדִי׃ יֶעֱרַב עָלָיו שִׂיחִי אָנֹכִי אֶשְׂמַח בַּד'׃ יִתַּמּוּ חַטָּאִים מִן־הָאָרֶץ וּרְשָׁעִים עוֹד אֵינָם בָּרֲכִי נַפְשִׁי אֶת־ד' הַלְלוּ־יָהּ׃

תהלים ק״ל

שִׁיר הַמַּעֲלוֹת מִמַּעֲמַקִּים קְרָאתִיךָ ד'׃ א' שִׁמְעָה בְקוֹלִי תִּהְיֶינָה אָזְנֶיךָ קַשֻּׁבוֹת לְקוֹל תַּחֲנוּנָי׃ אִם־עֲוֺנוֹת תִּשְׁמָר־קָהּ א' מִי יַעֲמֹד׃ כִּי־עִמְּךָ הַסְּלִיחָה לְמַעַן תִּוָּרֵא׃ קִוִּיתִי ד' קִוְּתָה נַפְשִׁי וְלִדְבָרוֹ הוֹחָלְתִּי׃ נַפְשִׁי לַא' מִשֹּׁמְרִים לַבֹּקֶר שֹׁמְרִים לַבֹּקֶר׃ יַחֵל יִשְׂרָאֵל אֶל־ד' כִּי־עִם־ד' הַחֶסֶד וְהַרְבֵּה עִמּוֹ פְדוּת׃ וְהוּא יִפְדֶּה אֶת־יִשְׂרָאֵל מִכֹּל עֲוֺנֹתָיו׃

כשיוצא מבית הקברות נוהגים לתלוש מעט עשב ולזרוק אותו אחריו, ואומרים:
