סדר ברית מילה

כשמביאים את התינוק, הקהל אומר:

בָּרוּךְ הַבָּא.

המוהל אומר:

וַיְדַבֵּר ד' אֶל־מֹשֶׁה לֵּאמֹר׃ פִּינְחָס בֶּן־אֶלְעָזָר בֶּן־אַהֲרֹן הַכֹּהֵן הֵשִׁיב אֶת־חֲמָתִי מֵעַל בְּנֵי־יִשְׂרָאֵל, בְּקַנְאוֹ אֶת־קִנְאָתִי בְּתוֹכָם, וְלֹא־כִלִּֽיתִי אֶת־בְּנֵי־יִשְׂרָאֵל בְּקִנְאָתִי׃ לָכֵן אֱמֹר, הִנְנִי נֹתֵן לוֹ אֶת־בְּרִיתִי שָׁלוֹם׃

המוהל: אַשְׁרֵי תִּבְחַר וּתְקָרֵב, יִשְׁכֹּן חֲצֵרֶֽיךָ

הקהל: נִשְׂבְּעָה בְּטוּב בֵּיתֶֽךָ, קְדֹשׁ הֵיכָלֶֽךָ׃

בארץ ישראל נוהגים שאבי הבן אומר פסוקים אלה, והקהל חוזר אחריו:

אִם־אֶשְׁכָּחֵךְ יְרוּשָׁלִָֽם, תִּשְׁכַּח יְמִינִי׃ תִּדְבַּק לְשׁוֹנִי לְחִכִּי
אִם־לֹא אֶזְכְּרֵֽכִי, אִם־לֹא אַעֲלֶה אֶת־יְרוּשָׁלִַֽם עַל רֹאשׁ שִׂמְחָתִי׃

שְׁמַע יִשְׂרָאֵל, ד' אֱלֹקֵֽינוּ, ד' אֶחָד׃

ד' מֶֽלֶךְ, ד' מָלָךְ, ד' יִמְלֹךְ לְעוֹלָם וָעֶד.
אָנָּא ד' הוֹשִֽׁיעָה נָּא
אָנָּא ד' הַצְלִיחָה נָא׃

מניחים את התינוק על הכסא של אליהו, והמוהל אומר:

זֶה הַכִּסֵּא שֶׁל אֵלִיָּֽהוּ הַנָּבִיא זָכוּר לַטּוֹב.

לִישׁוּעָתְךָ קִוִּֽיתִי ד'׃ שִׂבַּֽרְתִּי לִישׁוּעָתְךָ ד', וּמִצְוֹתֶֽיךָ עָשִֽׂיתִי׃ אֵלִיָּֽהוּ מַלְאַךְ הַבְּרִית, הִנֵּה שֶׁלְּךָ לְפָנֶֽיךָ, עֲמֹד עַל יְמִינִי וְסָמְכֵֽנִי. שִׂבַּֽרְתִּי לִישׁוּעָתְךָ ד'׃ שָׂשׂ אָנֹכִי עַל־אִמְרָתֶֽךָ, כְּמוֹצֵא שָׁלָל רָב׃ שָׁלוֹם רָב לְאֹהֲבֵי תוֹרָתֶֽךָ, וְאֵין־לָֽמוֹ מִכְשׁוֹל׃ אַשְׁרֵי תִּבְחַר וּתְקָרֵב, יִשְׁכֹּן חֲצֵרֶֽיךָ

הקהל:

נִשְׂבְּעָה בְּטוּב בֵּיתֶֽךָ, קְדֹשׁ הֵיכָלֶֽךָ׃

המוהל מברך:

בָּרוּךְ אַתָּה ד' אֱלֹקֵֽינוּ מֶֽלֶךְ הָעוֹלָם
אֲשֶׁר קִדְּשָֽׁנוּ בְּמִצְוֹתָיו וְצִוָּֽנוּ עַל הַמִּילָה.

אבי הבן מברך:

בָּרוּךְ אַתָּה ד' אֱלֹקֵֽינוּ מֶֽלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָֽׁנוּ בְּמִצְוֹתָיו וְצִוָּֽנוּ לְהַכְנִיסוֹ בִּבְרִיתוֹ שֶׁל אַבְרָהָם אָבִֽינוּ.

בארץ ישראל אבי הבן מברך גם:

בָּרוּךְ אַתָּה ד' אֱלֹקֵֽינוּ מֶֽלֶךְ הָעוֹלָם
שֶׁהֶחֱיָֽנוּ וְקִיְּמָֽנוּ וְהִגִּיעָֽנוּ לַזְּמַן הַזֶּה.

הקהל עונה:

אָמֵן. כְּשֵׁם שֶׁנִּכְנַס לַבְּרִית, כֵּן יִכָּנֵס לְתוֹרָה וּלְחֻפָּה וּלְמַעֲשִׂים טוֹבִים.

המברך נוטל כוס של יין ומברך:

בָּרוּךְ אַתָּה ד' אֱלֹקֵֽינוּ מֶֽלֶךְ הָעוֹלָם, בּוֹרֵא פְּרִי הַגָּֽפֶן.

בָּרוּךְ אַתָּה ד' אֱלֹקֵֽינוּ מֶֽלֶךְ הָעוֹלָם, אֲשֶׁר קִדַּשׁ יְדִיד מִבֶּֽטֶן, וְחֹק בִּשְׁאֵרוֹ שָׂם, וְצֶאֱצָאָיו חָתַם בְּאוֹת בְּרִית קֹֽדֶשׁ. עַל כֵּן בִּשְׂכַר זֹאת, קֵל חַי חֶלְקֵֽנוּ צוּרֵֽנוּ צִוָּה לְהַצִּיל יְדִידוּת שְׁאֵרֵֽנוּ מִשַּֽׁחַת, לְמַֽעַן בְּרִיתוֹ אֲשֶׁר שָׂם בִּבְשָׂרֵֽנוּ. בָּרוּךְ אַתָּה ד', כּוֹרֵת הַבְּרִית.

כשאומר ״בדמיך חיי״ נותנים לתינוק מעט מן היין.

אֱלֹקֵֽינוּ וֵאלֹקֵי אֲבוֹתֵֽינוּ, קַיֵּם אֶת הַיֶּֽלֶד הַזֶּה לְאָבִיו וּלְאִמּוֹ, וְיִקָּרֵא שְׁמוֹ בְּיִשְׂרָאֵל (פלוני בֶּן פלוני). יִשְׂמַח הָאָב בְּיוֹצֵא חֲלָצָיו וְתָגֵל אִמּוֹ בִּפְרִי בִטְנָהּ, כַּכָּתוּב: יִשְׂמַח־אָבִֽיךָ וְאִמֶּֽךָ, וְתָגֵל יוֹלַדְתֶּֽךָ׃ וְנֶאֱמַר: וָאֶעֱבֹר עָלַֽיִךְ וָאֶרְאֵךְ מִתְבּוֹסֶֽסֶת בְּדָמָֽיִךְ, וָאֹֽמַר לָךְ בְּדָמַֽיִךְ חֲיִי, וָאֹֽמַר לָךְ בְּדָמַֽיִךְ חֲיִי׃

וְנֶאֱמַר: זָכַר לְעוֹלָם בְּרִיתוֹ, דָּבָר צִוָּה לְאֶֽלֶף דּוֹר׃ אֲשֶׁר כָּרַת אֶת־אַבְרָהָם, וּשְׁבוּעָתוֹ לְיִשְׂחָק׃ וַיַּעֲמִידֶֽהָ לְיַעֲקֹב לְחֹק, לְיִשְׂרָאֵל בְּרִית עוֹלָם׃ וְנֶאֱמַר: וַיָּֽמָל אַבְרָהָם אֶת־יִצְחָק בְּנוֹ בֶּן־שְׁמֹנַת יָמִים, כַּאֲשֶׁר צִוָּה אֹתוֹ אֱלֹקִים׃ הוֹדוּ לַד' כִּי־טוֹב, כִּי לְעוֹלָם חַסְדּוֹ׃

הקהל אומר, והמברך חוזר:

הוֹדוּ לַד' כִּי־טוֹב, כִּי לְעוֹלָם חַסְדּוֹ׃

(פלוני בֶּן פלוני) זֶה הַקָּטֹן גָּדוֹל יִהְיֶה, כְּשֵׁם שֶׁנִּכְנַס לַבְּרִית, כֵּן יִכָּנֵס לְתוֹרָה וּלְחֻפָּה וּלְמַעֲשִׂים טוֹבִים.

המוהל אומר:

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב הוּא יְבָרֵךְ אֶת הַיֶּלֶד רַךְ הַנִּמּוֹל (פלוני בן פלוני) וְיִשְׁלַח לוֹ רְפוּאָה שְׁלֵמָה, בַּעֲבוּר שֶׁנִּכְנַס לַבְּרִית. וּכְשֵׁם שֶׁנִּכְנַס לַבְּרִית כֵּן יִכָּנֵס לְתוֹרָה וּלְחֻפָּה וּלְמַעֲשִׂים טוֹבִים. וְנֹאמַר אָמֵן.

אבי הבן אומר:

רִבּוֹנוֹ שֶׁל עוֹלָם, יְהִי רָצוֹן מִלְּפָנֶיךָ שֶׁיְּהֵא חָשׁוּב וּמְרֻצֶּה וּמְקֻבָּל לְפָנֶיךָ כְּאִלּוּ הִקְרַבְתִּיהוּ לִפְנֵי כִסֵּא כְבוֹדֶךָ. וְאַתָּה בְּרַחֲמֶיךָ הָרַבִּים שְׁלַח עַל יְדֵי מַלְאָכֶיךָ הַקְּדוֹשִׁים נְשָׁמָה קְדוֹשָׁה וּטְהוֹרָה לִבְנִי הַנִּמּוֹל עַתָּה לְשִׁמְךָ הַגָּדוֹל, וְשֶׁיִּהְיֶה לִבּוֹ פָּתוּחַ כְּפִתְחוֹ שֶׁל אוּלָם בְּתוֹרָתְךָ הַקְּדוֹשָׁה לִלְמֹד וּלְלַמֵּד לִשְׁמֹר וְלַעֲשׂוֹת. וְתֵן לוֹ אֲרִיכוּת יָמִים וְשָׁנִים, חַיִּים שֶׁל יִרְאַת חֵטְא, חַיִּים שֶׁל עֹשֶׁר וְכָבוֹד, חַיִּים שֶׁתְּמַלֵּא מִשְׁאֲלוֹת לִבּוֹ לְטוֹבָה. אָמֵן וְכֵן יְהִי רָצוֹן.
